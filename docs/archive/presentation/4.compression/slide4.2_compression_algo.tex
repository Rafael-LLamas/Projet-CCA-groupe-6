\begin{frame}{Compression}
  \begin{algorithm}[H]
    \caption{Generator Compression}
    \begin{algorithmic}[1]
      \Function{gr\_mat\_generator\_compress\_aux}{$G, H$}
      \State $P, L, U \gets \text{LU\_Decomposition}(G^T)$
      \Comment{$\mathcal{O}(\alpha^2 n)$}
      \State $r \gets \text{Number of non-zero rows in } U$
      \Comment{$\mathcal{O}(\alpha)$}
      \If{$r < 1$}
      \State \textbf{return} Generators of a $0$ matrix
      \Comment{$\mathcal{O}(1)$}
      \EndIf
      \State $G_{D} \gets (U_{[0:r-1],*})^T$ \Comment{$\mathcal{O}(\alpha n)$}
      \State $H_{D} \gets H P^T \times L_{*,[0:r-1]}$
      \Comment{$\mathcal{O}(\alpha^2 n)$}
      \State \textbf{return} $G_D, H_D$ \Comment{$\mathcal{O}(1)$}
      \EndFunction
    \end{algorithmic}
  \end{algorithm}

  Giving us a total complexity of $\mathcal{O}(\alpha^2 n)$
\end{frame}
