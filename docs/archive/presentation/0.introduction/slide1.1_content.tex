\begin{frame}{Toeplitz Matrices}
  \begin{definition}
    A matrix $A \in \mathbb{K}^{n \times m}$ is \textbf{Toeplitz} if
    $A_{i,j} = A_{i+1,j+1}$ for all valid $i, j$.
  \end{definition}

  {\footnotesize
    \[
      A =
      \begin{pmatrix}
        a_0     & a_{-1} & \cdots   &        &        &       & & a_{-(m-1)} \\
        a_1     & a_0    & a_{-1}   &        &        &       & & a_{1-(m-1)} \\
        \vdots  & a_1    & a_0      & \ddots &        &       & & a_{2-(m-1)} \\
        \vdots  &        & \ddots   & \ddots & \ddots &       & & \vdots     \\
        a_{n-1} & \cdots & \cdots   & a_1    & a_0    & a_{-1}&\cdots&
        a_{(n-1)-(m-1)}
      \end{pmatrix}
    \]
  }
  \vspace{-0.5em}
  \[ A_{i,j} = a_{i-j}, \quad 0 \le i < n,\; 0 \le j < m \]

  \begin{block}{Examples of Toeplitz Matrices}
    \centering
    $A_{1} =
    \begin{pmatrix}
      4 & 3 & 2 & 1 \\
      5 & 4 & 3 & 2 \\
      6 & 5 & 4 & 3 \\
      7 & 6 & 5 & 4
    \end{pmatrix}$
    \quad
    $A_{2} =
    \begin{pmatrix}
      120 & 432 & 584 & 394 & 159 \\
      312 & 120 & 432 & 584 & 394 \\
      985 & 312 & 120 & 432 & 584
    \end{pmatrix}$
  \end{block}
\end{frame}
