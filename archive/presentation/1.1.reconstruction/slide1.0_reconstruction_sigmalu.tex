    \begin{column}{0.5\textwidth}
      \begin{block}{Reconstruction}
        \[
          A = \sum_{k=1}^{\alpha} L(g_k)\, U(h_k)
        \]
      \end{block}
    \end{column}
  \end{columns}

  \vspace{1cm}

  Defining $L$ and $U$ are Lower and Upper matrices respectively:
  $$
  L(g_k) =
  \begin{bmatrix}
    g_{k,1} & 0 & 0 & 0 \\
    g_{k,2} & g_{k,1} & 0 & 0 \\
    g_{k,3} & g_{k,2} & g_{k,1} & 0 \\
    g_{k,4} & g_{k,3} & g_{k,2} & g_{k,1}
  \end{bmatrix}
  \qquad
  U(h_k) =
  \begin{bmatrix}
    h_{k,1} & h_{k,2} & h_{k,3} & h_{k,4} \\
    0 & h_{k,1} & h_{k,2} & h_{k,3} \\
    0 & 0 & h_{k,1} & h_{k,2} \\
    0 & 0 & 0 & h_{k,1}
  \end{bmatrix}
  $$

\end{frame}
