\begin{frame}{Multiplication of Toeplitz-like Matrices}
  \begin{block}{Generator Multiplication}
    \begin{align*}
      A &\simeq (T, U), & B &\simeq (G, H) \\
      W &= Z A Z^\top G, & V &= B^\top U \\
      \text{LastCol} &= (0, \dots, 0, 1)^\top, & a &= Z A \, \text{LastCol} \\
      b &= Z B^\top \text{LastCol}, & C &\simeq (T \mid W \mid a, \,
      V \mid H \mid b)
    \end{align*}
  \end{block}

  \vspace{1em}

  \begin{block}{Complexities}
    Naive: $\mathcal{O}(n^3)$ \\
    Concatenate: $\mathcal{O}(\alpha^2 M(n))$
  \end{block}
\end{frame}
