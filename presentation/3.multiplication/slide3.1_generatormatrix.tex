\begin{frame}{Multiplication of Toeplitz-like Matrices}
  \begin{columns}
    \begin{column}{0.5\textwidth}
  \begin{block}{Generator Multiplication}
        \begin{align*}
          A &= (T,U),\qquad B &= (G,H) \\
          W &= Z A Z^t G \\
          V &= B^t U \\
          \alpha &= Z A \\
          \beta &= Z B^t \\
          C &=(T |W| \alpha, V|H | \beta)
        \end{align*}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{alertblock}{Notice}
    from $O(n^3)$ to $O(\alpha^2 M(n))$.
  \end{alertblock}
\end{frame}
