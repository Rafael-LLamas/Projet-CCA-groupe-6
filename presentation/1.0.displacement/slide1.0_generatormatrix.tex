    \begin{column}{0.5\textwidth}

      \begin{block}{Displacement Operator}
        \[
          \nabla A = A - Z A Z^T = G H^T
        \]
        For Toeplitz: $\mathrm{rank}(\nabla A) \leq 2$
      \end{block}

    \end{column}

  % \begin{alertblock}{Notice}

  %   $O(n^2)$ to $O(\alpha n)$.

  \begin{block}{Shift Matrix}
    \centering
    $Z =
    \begin{pmatrix}
      0       &           &   &  \\
      1       &           &   &  \\
      & \ddots    &   &  \\
      &           & 1 & 0
    \end{pmatrix}$
    \quad
    $Z^{T} =
    \begin{pmatrix}
      0       & 1         &       &  \\
      &           &\ddots &  \\
      &           &       & 1\\
      &           &       & 0
    \end{pmatrix}$
  \end{block}
