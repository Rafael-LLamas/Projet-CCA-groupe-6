\begin{frame}{Addition of Toeplitz-like Matrices}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Generator Addition}
        \[
           A = (T,U), B = (G,H)
            \implies A + B = (T | G, U | H)
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{alertblock}{Notice}
    from $O(n^2)$ to $O(\alpha M(n))$.
  \end{alertblock}
\end{frame}
