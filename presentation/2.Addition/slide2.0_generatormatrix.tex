\begin{frame}{Addition of Toeplitz-like Matrices}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{block}{Generator Addition}
        \[
          A \simeq (T,U), B \simeq (G,H) \Rightarrow A + B \simeq (T | G, U | H)
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{block}{Complexities}
    Naive: $O(n^2)$           \\
    Concatenation : $O(\alpha n)$
  \end{block}
\end{frame}
