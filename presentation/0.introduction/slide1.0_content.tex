\begin{frame}{Context}
  \begin{itemize}
    \item Operations on large structured matrices are often limited by memory bandwith.
    \item Standart operatons on $n^2$ elements lead to unnecessary computations.
  \end{itemize}
  \vspace{1em}
  \begin{block}{Goal}
    Optimize memory footprint and execution time on structured matrices by implementing displacement operators and generator-based representations.
  \end{block}
\end{frame}
