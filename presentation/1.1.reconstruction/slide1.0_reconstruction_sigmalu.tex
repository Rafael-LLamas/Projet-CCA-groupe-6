      \begin{block}{Reconstruction}
        \[
          A = \sum_{k=1}^{\alpha} L(g_k)\, U(h_k)
        \]
      \end{block}
    \end{column}
  \end{columns}

  \vspace{1cm}
  
  Defining $L$ and $U$ are Lower and Upper matrices respectively:
  \[
      L(g_k)=
      \begin{bmatrix}
          g_1 & 0 & 0 & 0 \\
          g_2 & g_1 & 0 & 0 \\
          g_3 & g_2 & g_1 & 0 \\
          g_4 & g_3 & g_2 & g_1
      \end{bmatrix}
      \qquad
      U(h_k)=
      \begin{bmatrix}
          h_1 & h_2 & h_3 & h_4 \\
          0 & h_1 & h_2 & h_3 \\
          0 & 0 & h_1 & h_2 \\
          0 & 0 & 0 & h_1
      \end{bmatrix}
  \]
  
\end{frame}
