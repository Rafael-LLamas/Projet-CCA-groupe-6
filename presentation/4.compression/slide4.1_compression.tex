\begin{frame}{Compression}

    \begin{block}{Definition}
        Matrices $G \in \mathbb{R}^{n \times \alpha}$ and $H \in \mathbb{R}^{n \times \alpha}$ represent a matrix $\nabla A = G H^T$.
        \vspace{0.5em}
        
        \textbf{Goal:} Find two matrices $G' \in \mathbb{R}^{n \times r}$ and $H' \in \mathbb{R}^{n \times r}$ such that:
        \[
            G'H'^T = G H^T
        \]
    \end{block}

\end{frame}
