\begin{frame}{Compression}

    \begin{block}{Definition}
        Matrices $G \in \mathbb{R}^{n \times \alpha}$ and $H \in \mathbb{R}^{n \times \alpha}$ representing a displacement matrix $\nabla A = G H^T$, where the actual rank of $\nabla A$ is $r < \alpha$.
        \vspace{0.5em}
        
        \textbf{Goal:} Find generators $G' \in \mathbb{R}^{n \times r}$ and $H' \in \mathbb{R}^{n \times r}$ s.t. the product is the same:
        \[
            G'H'^T = G H^T
        \]
    \end{block}

\end{frame}
